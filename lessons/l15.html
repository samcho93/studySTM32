PART 4 프로젝트 · 약 120분 · 중급

L15미니 프로젝트: 스마트 선풍기

가변저항 → PWM 모터 속도, 버튼 모드 전환, LCD 표시, UART 로그를 하나의 프로그램으로 묶습니다.

선수 레슨: L09 L10 L13
시뮬레이터에서 실습HAL 레퍼런스

학습 목표#

  • 요구사항을 입력 · 출력 · 상태(모드) 로 나누고, 핀 배치와 CubeMX 설정표로 옮깁니다.
  • ADC · PWM · EXTI · I2C LCD · UART 를 하나의 프로그램으로 묶고, HAL_GetTick() 으로 작업 주기를 나눕니다.
  • 인터럽트(버튼)와 메인 루프 사이에서 volatile 플래그로 상태를 안전하게 주고받습니다.
  • 동작 시험 체크리스트로 스스로 검증합니다.

요구사항#

번호요구사항
R1가변저항을 돌리면 선풍기(DC 모터) 속도가 0 ~ 100 % 로 바뀐다 (MANUAL 모드)
R2B1 버튼을 누를 때마다 모드가 OFF → MANUAL → BREEZE(자연풍) → OFF 순서로 바뀐다
R3BREEZE 모드는 8초 주기로 30 % ~ 80 % 사이를 부드럽게 오르내린다
R4LCD 1행에 모드, 2행에 현재 속도(%)를 표시한다 (0.2초마다 갱신)
R5UART 로 1초마다 [시간] mode=… pot=… duty=…% 로그를 보낸다
R6모드가 OFF 가 아니면 LD2 가 켜진다

프로그램 구조#

text
            ┌────────── B1 인터럽트 (EXTI) ───────────┐
            │  mode = (mode + 1) % 3,  mode_changed = 1 │
            └──────────────────────────────────────────┘
main 루프 (20 ms 마다)
  1. pot  = read_pot()                 ← ADC
  2. duty = compute_duty()             ← 모드별 계산
  3. motor_set(duty), LD2              ← PWM / GPIO
  4. 0.2초마다(또는 모드 변경 시) LCD 갱신   ← I2C
  5. 1초마다 UART 로그                    ← printf

느린 일(LCD, UART)은 주기를 길게, 빠른 일(모터 갱신)은 매번 하는 것이 핵심입니다. LCD 를 20 ms 마다 쓰면 I2C 전송에 시간이 많이 들어 루프가 느려집니다. 각 작업의 "마지막 실행 시각" 을 저장해 두고 HAL_GetTick() 과 비교합니다(L06, L07).

구현#

시뮬레이터에서 열기스마트 선풍기 열기새 창l15-fan

전체 코드입니다. LCD 드라이버(L10)와 motor_set(L13)은 앞 레슨의 것을 그대로 가져왔습니다.

cCore/Src/main.c
/* USER CODE BEGIN Includes */
#include <stdio.h>
/* USER CODE END Includes */

/* USER CODE BEGIN PV */
#define MODE_OFF     0
#define MODE_MANUAL  1
#define MODE_BREEZE  2

const char *MODE_NAME[3] = {"OFF", "MANUAL", "BREEZE"};
volatile uint8_t mode = MODE_OFF;
volatile uint8_t mode_changed = 1;
volatile uint32_t last_press = 0;
uint32_t pot = 0;
int duty = 0;                        // 0 ~ 999
uint32_t last_lcd = 0;
uint32_t last_log = 0;
char line[17];
/* USER CODE END PV */

/* USER CODE BEGIN PFP */
int __io_putchar(int ch);
/* USER CODE END PFP */

/* USER CODE BEGIN 0 */
/* ---- I2C LCD1602 (PCF8574 백팩) 드라이버 ----
 * PCF8574 비트: P0=RS, P1=RW, P2=EN, P3=백라이트, P4~P7=D4~D7 (4비트 모드) */
#define LCD_ADDR  (0x27 << 1)   // HAL 은 7비트 주소를 왼쪽으로 1비트 민 값을 받습니다
#define LCD_RS    0x01
#define LCD_EN    0x04
#define LCD_BL    0x08

/* 상위 4비트(nibble)를 EN 펄스와 함께 보냅니다. */
void lcd_send4(uint8_t nibble, uint8_t rs)
{
  uint8_t data[2];
  data[0] = nibble | LCD_BL | LCD_EN | rs;   // EN = 1
  data[1] = nibble | LCD_BL | rs;            // EN = 0 (하강 에지에서 LCD 가 읽음)
  HAL_I2C_Master_Transmit(&hi2c1, LCD_ADDR, data, 2, 10);
}

/* 8비트 값을 상위 -> 하위 4비트 순서로 두 번 나눠 보냅니다. */
void lcd_send(uint8_t value, uint8_t rs)
{
  lcd_send4(value & 0xF0, rs);
  lcd_send4((value << 4) & 0xF0, rs);
}

void lcd_cmd(uint8_t cmd)
{
  lcd_send(cmd, 0);
  if (cmd == 0x01 || cmd == 0x02) HAL_Delay(2);   // Clear / Home 은 1.5 ms 이상 걸림
}

void lcd_data(uint8_t ch)
{
  lcd_send(ch, LCD_RS);
}

void lcd_init(void)
{
  HAL_Delay(50);                 // 전원 인가 후 안정화
  lcd_send4(0x30, 0); HAL_Delay(5);
  lcd_send4(0x30, 0); HAL_Delay(1);
  lcd_send4(0x30, 0); HAL_Delay(1);
  lcd_send4(0x20, 0); HAL_Delay(1);   // 4비트 모드로 전환
  lcd_cmd(0x28);                 // 4비트, 2줄, 5x8 글꼴
  lcd_cmd(0x08);                 // 표시 끔
  lcd_cmd(0x01);                 // 화면 지우기
  lcd_cmd(0x06);                 // 글자 쓰면 커서 오른쪽으로
  lcd_cmd(0x0C);                 // 표시 켬, 커서 숨김
}

void lcd_set_cursor(uint8_t row, uint8_t col)
{
  uint8_t addr = (row == 0 ? 0x00 : 0x40) + col;   // 1행 DDRAM 0x00, 2행 0x40
  lcd_cmd(0x80 | addr);
}

void lcd_print(const char *s)
{
  while (*s) {
    lcd_data((uint8_t)*s++);
  }
}

/* speed: -999 ~ +999. 부호가 방향, 크기가 PWM 듀티(ARR=999 기준)입니다. */
void motor_set(int speed)
{
  if (speed > 999) speed = 999;
  if (speed < -999) speed = -999;

  if (speed > 0) {                 // 정방향: IN1=1, IN2=0
    HAL_GPIO_WritePin(MOTOR_IN1_GPIO_Port, MOTOR_IN1_Pin, GPIO_PIN_SET);
    HAL_GPIO_WritePin(MOTOR_IN2_GPIO_Port, MOTOR_IN2_Pin, GPIO_PIN_RESET);
  } else if (speed < 0) {          // 역방향: IN1=0, IN2=1
    HAL_GPIO_WritePin(MOTOR_IN1_GPIO_Port, MOTOR_IN1_Pin, GPIO_PIN_RESET);
    HAL_GPIO_WritePin(MOTOR_IN2_GPIO_Port, MOTOR_IN2_Pin, GPIO_PIN_SET);
    speed = -speed;
  } else {                         // 정지: EN=0 이면 자유 회전으로 멈춤
    HAL_GPIO_WritePin(MOTOR_IN1_GPIO_Port, MOTOR_IN1_Pin, GPIO_PIN_RESET);
    HAL_GPIO_WritePin(MOTOR_IN2_GPIO_Port, MOTOR_IN2_Pin, GPIO_PIN_RESET);
  }
  __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_1, speed);   // ENA 에 PWM
}

uint32_t read_pot(void)
{
  uint32_t v = 0;
  HAL_ADC_Start(&hadc1);
  if (HAL_ADC_PollForConversion(&hadc1, 10) == HAL_OK) {
    v = HAL_ADC_GetValue(&hadc1);
  }
  HAL_ADC_Stop(&hadc1);
  return v;
}

/* 모드에 따라 목표 듀티(0~999)를 계산합니다. */
int compute_duty(void)
{
  if (mode == MODE_MANUAL) {
    return (int)(pot * 999 / 4095);
  }
  if (mode == MODE_BREEZE) {                    // 자연풍: 8초 주기로 30 % ~ 80 % 오르내림
    uint32_t t = HAL_GetTick() % 8000;
    uint32_t tri = (t < 4000) ? t : 8000 - t;   // 0 -> 4000 -> 0 삼각파
    return 300 + (int)(tri * 500 / 4000);
  }
  return 0;                                     // MODE_OFF
}
/* USER CODE END 0 */

  /* USER CODE BEGIN 2 */
  HAL_TIM_PWM_Start(&htim3, TIM_CHANNEL_1);
  lcd_init();
  printf("\r\nSmart fan ready. B1 = mode\r\n");
  /* USER CODE END 2 */

  /* USER CODE BEGIN WHILE */
  while (1)
  {
    /* USER CODE END WHILE */

    /* USER CODE BEGIN 3 */
    pot = read_pot();
    duty = compute_duty();
    motor_set(duty);
    HAL_GPIO_WritePin(LD2_GPIO_Port, LD2_Pin, mode != MODE_OFF ? GPIO_PIN_SET : GPIO_PIN_RESET);

    if (mode_changed || HAL_GetTick() - last_lcd >= 200) {   // LCD 는 0.2초마다만 갱신
      mode_changed = 0;
      last_lcd = HAL_GetTick();
      sprintf(line, "Mode: %-10s", MODE_NAME[mode]);
      lcd_set_cursor(0, 0);
      lcd_print(line);
      sprintf(line, "Speed: %3d %%    ", duty * 100 / 999);
      lcd_set_cursor(1, 0);
      lcd_print(line);
    }

    if (HAL_GetTick() - last_log >= 1000) {                  // UART 로그는 1초마다
      last_log = HAL_GetTick();
      printf("[%6lu] mode=%-6s pot=%4lu duty=%3d%%\r\n",
             last_log, MODE_NAME[mode], pot, duty * 100 / 999);
    }
    HAL_Delay(20);
  }
  /* USER CODE END 3 */

/* USER CODE BEGIN 4 */
/* printf 리타깃: syscalls.c 의 _write() 가 글자마다 이 함수를 부릅니다. */
int __io_putchar(int ch)
{
  HAL_UART_Transmit(&huart2, (uint8_t *)&ch, 1, HAL_MAX_DELAY);
  return ch;
}

void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin)
{
  if (GPIO_Pin == B1_Pin && HAL_GetTick() - last_press > 200) {
    last_press = HAL_GetTick();
    mode = (mode + 1) % 3;           // OFF -> MANUAL -> BREEZE -> OFF
    mode_changed = 1;
  }
}
/* USER CODE END 4 */

코드 읽기#

  • 모드 상수 — #define MODE_OFF 0 처럼 이름을 붙이면 if (mode == 1) 보다 읽기 쉽고 실수가 줄어듭니다.
  • **const char *MODE_NAME[3]** — 모드 번호를 인덱스로 문자열을 꺼냅니다. LCD 와 UART 가 같은 이름을 씁니다.
  • **compute_duty()** — 모드별 계산을 한 함수로 모아, 새 모드를 추가할 때 이 함수만 고치면 됩니다.

BREEZE 모드의 삼각파는 HAL_GetTick() % 8000 으로 8초 주기 시간을 만들고, 앞 4초는 증가, 뒤 4초는 감소하게 했습니다.

  • **"Mode: %-10s"** — 6글자 + 10칸 = 16글자로 LCD 한 줄을 꽉 채워 이전 글자를 지웁니다. "Speed: %3d %% " 도 16글자입니다.
  • **%%** — printf 에서 % 글자 자체를 출력합니다.
  • 인터럽트는 짧게 — 콜백은 디바운스 후 mode 와 mode_changed 만 바꾸고, LCD 갱신은 메인 루프가 합니다.

시험#

동작 시험 체크리스트
스스로 확인
  • LCD 갱신을 매 루프(20 ms)마다 하면 무엇이 나빠질까요? 대략 몇 ms 가 걸리는지 계산해 보세요(100 kHz I2C, 글자당 4바이트 + 주소).
  • mode 에 volatile 이 없으면 실물에서 어떤 증상이 생길 수 있나요?

자주 나는 오류와 해결#

증상원인해결
LCD 가 깜빡이거나 글자가 겹침매 루프 Clear(0x01) 사용, 16칸을 다 채우지 않음Clear 대신 16칸 덮어쓰기
버튼 반응이 한 박자 늦음루프 안의 긴 작업(LCD 를 매번 갱신)LCD·UART 주기 분리, mode_changed 로 즉시 갱신
MANUAL 에서 가변저항을 0 으로 해도 모터가 윙윙거림낮은 듀티에서 기동하지 못함데드존(L13) 또는 최소 듀티
모드가 두 칸씩 넘어감디바운스 없음콜백의 200 ms 시간 비교
빌드 오류 MODE_NAME 관련 경고문자열 배열 선언 위치PV 구역에 const char *MODE_NAME[3]
실물에서 모터가 돌면 LCD 가 깨짐모터 노이즈·전원 강하가 I2C 에 영향모터 전원 분리, 콘덴서, 배선 분리

과제#

  1. 자동 모드(AUTO) 를 추가하세요: 조도센서(CDS, PA1 ADC1_IN1, L09) 값이 어두우면 약하게, 밝으면 강하게. compute_duty() 와 MODE_NAME 만 고치면 되도록 설계합니다.
  2. 꺼짐 예약: UART 명령 timer 10 을 받으면 10초 뒤 OFF 로 바뀌고, LCD 2행 오른쪽에 남은 초를 표시하세요.
  3. 모드가 바뀔 때 속도가 갑자기 변하지 않도록, 목표 듀티까지 초당 30 % 씩 천천히 따라가는 램프 기능을 넣으세요.
  4. 설계 문서를 작성하세요: 요구사항표, 핀표, 설정표, 구조도, 시험 결과(체크리스트 캡처).
